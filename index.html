Code Example

Task Codewars (Odd-Even String Sort): Given a string, turn each character of even-indexed positions and odd-indexed positions into two separate strings, then return them as a single string separated by a space.

function sortMyString(a) {
    let odd = '';
    let even = '';
    for (let i = 0; i < a.length; i++) {
        i % 2 ? odd += a[i] : even += a[i];
    }
    return `${even} ${odd}`;
}
